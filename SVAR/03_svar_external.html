read_fred <- function(file, name) {
  read_csv(here("data", "raw", file), show_col_types = FALSE) |>
    rename(date = 1, value = 2) |>
    mutate(date = as.Date(date), value = as.numeric(value)) |>
    filter(!is.na(value)) |>
    mutate(date = floor_date(date, "quarter")) |>
    group_by(date) |>
    summarise(price = mean(value), n = n(), .groups = "drop") |>
    filter(n == 3) |>
    transmute(date, !!name := 400 * (log(price) - log(lag(price))))   # annualized quarterly growth
}

Step 4: Decompose inflation into core, non-core, and commodity contributions

A1   <- Acoef(varx)[[1]]
cf   <- sapply(varx$varresult, coef)                    # rows: coefficients, columns: equations
Cx   <- t(cf[colnames(X), ])                            # 2 x 4: effects of the commodity variables
xbar <- colMeans(X)
mu   <- as.numeric(solve(diag(2) - A1, cf["const", ] + Cx %*% xbar))   # long-run means
eps  <- resid(varx) %*% t(solve(B)); colnames(eps) <- c("noncore", "core")
xd   <- sweep(X[-1, , drop = FALSE], 2, xbar)           # commodity deviations from average, aligned with residuals

n <- nrow(eps)
core <- nc <- ext <- matrix(0, n, 2)
for (t in seq_len(n)) {
  prev <- function(m) if (t == 1) c(0, 0) else m[t - 1, ]
  core[t, ] <- A1 %*% prev(core) + B[, "core"]    * eps[t, "core"]
  nc[t, ]   <- A1 %*% prev(nc)   + B[, "noncore"] * eps[t, "noncore"]
  ext[t, ]  <- A1 %*% prev(ext)  + Cx %*% xd[t, ]
}

qvx <- tibble(date          = dx$date[-1],
              infl          = dx$infl[-1],
              core          = mu[2] + core[, 2],
              noncore_part  = nc[, 2],
              external_part = ext[, 2])

# check: should start nonzero and die out
qvx |> mutate(gap = infl - (core + noncore_part + external_part)) |>
