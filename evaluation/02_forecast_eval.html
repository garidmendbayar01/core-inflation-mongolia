# forecast pi(t+h) from predictors at t, re-estimated at each origin using only past data
direct_fc <- function(df, h, predictors, origins, offset_pi = FALSE) {
  df  <- df |> arrange(date) |> mutate(y_lead = lead(pi, h))
  rhs <- if (offset_pi) c(predictors, "offset(pi)") else predictors     # offset: coefficient on pi fixed at 1
  map_dfr(origins, \(o) {
    train <- df |>
      filter(date <= o %m-% months(3 * h),
             if_all(all_of(c("y_lead", "pi", predictors)), \(x) !is.na(x)))
    fit <- lm(reformulate(rhs, "y_lead"), data = train)
    tibble(origin = o, h = h, target_date = o %m+% months(3 * h),
           forecast = as.numeric(predict(fit, filter(df, date == o))))
  })
}

# forecasts from one core measure: (1) core level as the forecast, (2) gap regression
core_fc <- function(core_q, name) {
  d <- fc_data |> left_join(core_q, by = "date") |> mutate(gap = core - pi)
  map_dfr(horizons, \(h) {
    o <- origins_h(h)
    bind_rows(
      d |> filter(date %in% o) |>
        transmute(origin = date, h = h, target_date = date %m+% months(3 * h),
                  forecast = core, model = paste(name, "(level)")),
      direct_fc(d, h, "gap", o, offset_pi = TRUE) |> mutate(model = paste(name, "(gap)"))
    )
  }) |>
    left_join(select(target, target_date = date, actual = pi), by = "target_date")
}

# forecasts from real-time vintages: each origin uses its own vintage of the core history
core_fc_rt <- function(vintages, name) {
  map_dfr(horizons, \(h) map_dfr(origins_h(h), \(o) {
    d <- fc_data |>
      filter(date <= o) |>
      left_join(vintages[[as.character(o)]], by = "date") |>
      mutate(gap = core - pi, y_lead = lead(pi, h))
    tr      <- d |> filter(date <= o %m-% months(3 * h), !is.na(y_lead), !is.na(gap))
    now     <- filter(d, date == o)
    fit_gap <- lm(y_lead ~ gap + offset(pi), data = tr)     # gap regression (Cogley, 2002)
    fit_reg <- lm(y_lead ~ core, data = tr)                 # core regression: nests the historical mean
    tibble(origin = o, h = h, target_date = o %m+% months(3 * h),
           model    = paste(name, c("(level)", "(gap)", "(regression)")),
           forecast = as.numeric(c(now$core, predict(fit_gap, now), predict(fit_reg, now))))
  })) |>
    left_join(select(target, target_date = date, actual = pi), by = "target_date")
}

gdp_lin <- readRDS(here("data", "svar", "gdp_linked_nsa.rds"))
cpi_qr  <- readRDS(here("data", "svar", "cpi_q_sa.rds")) |> select(date, cpi)

svar_vintage <- function(o) {
  g    <- gdp_lin |> filter(date <= o)
  cp   <- cpi_qr  |> filter(date <= o)
  g_sa <- final(seas(ts(g$gdp, start = c(2000, 1), frequency = 4)))
  c_sa <- final(seas(ts(cp$cpi, start = c(2006, 1), frequency = 4),
                     regression.aictest = NULL, transform.function = "log"))
  dat <- tibble(date = g$date,  growth = 400 * c(NA, diff(log(as.numeric(g_sa))))) |>
    inner_join(tibble(date = cp$date, infl = 400 * c(NA, diff(log(as.numeric(c_sa))))), by = "date") |>
    filter(!is.na(growth), !is.na(infl))

  vf  <- vars::VAR(as.matrix(dat[, c("growth", "infl")]), p = 1, type = "const")
  bq  <- vars::BQ(vf)
  B   <- bq$B %*% diag(c(sign(bq$LRIM[1, 1]), sign(bq$B[2, 2])))
  A1  <- vars::Acoef(vf)[[1]]
  mu  <- solve(diag(2) - A1, sapply(vf$varresult, \(m) coef(m)[["const"]]))
  eps <- resid(vf) %*% t(solve(B))

  contrib <- matrix(0, nrow(eps), 2)
  for (t in seq_len(nrow(eps))) {
    contrib[t, ] <- A1 %*% (if (t == 1) c(0, 0) else contrib[t - 1, ]) + B[, 2] * eps[t, 2]
  }
  tibble(date = dat$date[-1], core_q = mu[2] + contrib[, 2]) |>
    mutate(p_core = cumsum(core_q / 400),
           core   = 100 * (exp(p_core - lag(p_core, 4)) - 1)) |>
    filter(!is.na(core)) |>
    select(date, core)
}

# seasonally adjusted, annualized monthly inflation from one index (NA if X-13 fails)
sa_infl <- function(idx, start_date) {
  x  <- ts(idx, start = c(year(start_date), month(start_date)), frequency = 12)
  sa <- tryCatch(final(seas(x, regression.aictest = NULL, transform.function = "log")),
                 error = \(e) NULL)
  if (is.null(sa)) return(rep(NA_real_, length(idx) - 1))
  1200 * diff(log(as.numeric(sa)))
}

# Stock-Watson outlier rule: values more than 6 IQRs from the median -> median of previous 5 months
clean_outliers <- function(x) {
  bad <- abs(x - median(x, na.rm = TRUE)) > 6 * IQR(x, na.rm = TRUE)
  for (i in which(bad)) if (i > 1) x[i] <- median(x[max(1, i - 5):(i - 1)], na.rm = TRUE)
  x
}

dfm_vintage <- function(o) {
  infl <- div_m |>
    filter(date <= o %m+% months(2)) |>
    arrange(code, date) |>
    group_by(code) |>
    reframe(date = date[-1], pi = sa_infl(index, date[1]))
  wide <- infl |> pivot_wider(names_from = code, values_from = pi) |> arrange(date)

  X <- wide |> select(all_of(unname(dfm_codes))) |> mutate(across(everything(), clean_outliers))
  X <- X[, colSums(is.na(X)) == 0]                  # drop any division whose X-13 failed
  X <- scale(X)

  pc <- prcomp(X, center = FALSE, scale. = FALSE)
  f  <- pc$x[, 1] * sign(mean(pc$rotation[, 1]))    # sign: factor rises with inflation
  reg <- tibble(date = wide$date, pi = wide$headline, f = f)
  fit <- lm(pi ~ f, data = reg)

  reg |>
    mutate(dfm_m    = fitted(fit),
           core_yoy = 100 * (exp(zoo::rollmeanr(dfm_m, 12, fill = NA) / 100) - 1),
           date     = floor_date(date, "quarter")) |>
    group_by(date) |>
    summarise(core = mean(core_yoy), n = n(), .groups = "drop") |>
    filter(n == 3, !is.na(core)) |>
    select(-n)
}

Step 11: Diebold-Mariano and Clark-West tests

# Newey-West (Bartlett) long-run variance
hac_var <- function(x, lags) {
  n <- length(x); x <- x - mean(x)
  v <- sum(x^2) / n
  for (l in seq_len(lags)) v <- v + 2 * (1 - l / (lags + 1)) * sum(x[(l + 1):n] * x[1:(n - l)]) / n
  v
}

# Diebold-Mariano with Harvey-Leybourne-Newbold correction; positive = model beats benchmark
dm_test <- function(e_bench, e_model, h) {
  d <- e_bench^2 - e_model^2; n <- length(d)
  stat <- mean(d) / sqrt(hac_var(d, h - 1) / n) * sqrt((n + 1 - 2 * h + h * (h - 1) / n) / n)
  c(dm = stat, p_dm = 2 * pt(-abs(stat), df = n - 1))
}

# Clark-West for nested models; one-sided: does the larger model add information?
cw_test <- function(e_null, e_alt, f_null, f_alt, h) {
  adj <- e_null^2 - (e_alt^2 - (f_null - f_alt)^2); n <- length(adj)
  stat <- mean(adj) / sqrt(hac_var(adj, h - 1) / n)
  c(cw = stat, p_cw = pnorm(-stat))
}

aug_fc <- function(vintages, name) {
  map_dfr(horizons, \(h) map_dfr(origins_h(h), \(o) {
    d <- fc_data |>
      filter(date <= o) |>
      left_join(vintages[[as.character(o)]], by = "date") |>
      mutate(y_lead = lead(pi, h))
    tr  <- d |> filter(date <= o %m-% months(3 * h), !is.na(y_lead), !is.na(core))
    now <- filter(d, date == o)
    tibble(origin = o, h = h, target_date = o %m+% months(3 * h), measure = name,
           f_null = as.numeric(predict(lm(y_lead ~ pi,        data = tr), now)),
           f_alt  = as.numeric(predict(lm(y_lead ~ pi + core, data = tr), now)))
  })) |>
    left_join(select(target, target_date = date, actual = pi), by = "target_date")
}
