Seasonally adjust each division and compute monthly inflation

sa_one <- function(idx, code) {
  tryCatch(
    as.numeric(seasonal::final(seasonal::seas(ts(idx, start = c(2010, 1), frequency = 12)))),
    error = function(e) { message("X-13 failed for ", code, ", using unadjusted"); idx }
  )
}

div_sa <- div12 |>
  arrange(code, date) |>
  group_by(code) |>
  mutate(index_sa = sa_one(index, first(code)),
         pi_sa    = 1200 * log(index_sa / lag(index_sa))) |>
  ungroup() |>
  filter(!is.na(pi_sa))

Outlier adjustment (Stock-Watson rule) and standardization

sw_clean <- function(x, k = 6) {
  out <- abs(x - median(x)) > k * IQR(x)
  for (t in which(out)) {
    x[t] <- if (t > 1) median(x[max(1, t - 5):(t - 1)]) else median(x)
  }
  x
}

div_clean <- div11 |>
  group_by(code) |>
  mutate(pi_clean = sw_clean(pi_sa),
         outlier  = pi_clean != pi_sa,
         z        = (pi_clean - mean(pi_clean)) / sd(pi_clean)) |>
  ungroup()

div_clean |>
